    // Minimal markdown-to-HTML converter
    function markdownToHtml(md) {
      let html = md;
      html = html.replace(/^### (.*$)/gim, '
$1
');
      html = html.replace(/^## (.*$)/gim, '
$1
');
      html = html.replace(/^# (.*$)/gim, '
$1
');
      html = html.replace(/\*\*(.*?)\*\*/gim, '$1');
      html = html.replace(/\*(.*?)\*/gim, '$1');
      html = html.replace(/\n\n/gim, '

');
      html = html.replace(/\n/gim, '
');
      html = html.replace(/\[(.*?)\]\((.*?)\)/gim, '$1');
      html = html.replace(/^- (.*)$/gim, '
  • $1
    • ');
      html = html.replace(/
  • (.*?)<\/li>/gim, '
    • $1
    ');
      return html;
    }
